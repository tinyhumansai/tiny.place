def sign_admin_request(data: Dict) -> Dict:
    """Signs a request with the admin Ed25519 signing key."""
    signer = Signer.from_hex(ADMIN_SIGNING_KEY)
    signed_data = signer.sign(json.dumps(data).encode())
    return {
        **data,
        "signature": signed_data.decode(HexEncoder),
        "admin_key": signer.key.decode(HexEncoder)
    }

def approve_bounty(bounty_id: str, winner: str, reward: float) -> bool:
    """Attempts to approve a bounty payout via the API."""
    payload = {
        "bounty_id": bounty_id,
        "winner": winner,
        "reward": reward,
        "approved_at": datetime.utcnow().isoformat() + "Z"
    }
    signed_payload = sign_admin_request(payload)
    
    try:
        response = requests.post(f"{BOUNTIES_ENDPOINT}/{bounty_id}{APPROVE_ENDPOINT}", json=signed_payload)
        response.raise_for_status()  # Raise an exception for bad status codes
        print(f"Bounty {bounty_id} approved successfully.")
        return True
    except requests.exceptions.RequestException as e:
        print(f"Error approving bounty {bounty_id}: {e}")
        return False

def main():
    """Iterates over stuck bounties and attempts to approve each for payout."""
